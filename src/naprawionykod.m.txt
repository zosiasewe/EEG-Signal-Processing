%% ===============================================================
%  MASTER ANALYSIS merge, summarize, plot, and export (with polygon)
%  ===============================================================
timestamp_str = datestr(now, 'yyyy_mm_dd_HH_MM_SS');
fprintf('=== Batch Results Concatenation & Analysis ===n');
fprintf('Timestamp %snn', timestamp_str);

% Hide axes toolbar in exported images
try, set(groot,'defaultAxesToolbarVisible','off'); end %#okTRYNC

% ---------- Which sets ----------
batched_sets  = {'PopulationSizes','OffspringRatio','TestRatioComparison'};
complete_sets = {'TreesComparison','ExtractedFeatures','SelectedFeatures'};

all_results = {};

%% ---------------------------------------------------------------
% 1) Load COMPLETE sets (non-batched)
% ---------------------------------------------------------------
fprintf('Loading complete results...n');
for i = 1numel(complete_sets)
    set_name = complete_sets{i};
    files = dir(sprintf('results_%s_.mat', set_name));
    files = files(~contains({files.name}, 'batch')); % exclude batches
    if ~isempty(files)
        [~, idx] = max([files.datenum]);
        data = load(files(idx).name);
        S = data.set_results;
        if ~isfield(S,'name')  isempty(S.name), S.name = set_name; end
        all_results{end+1} = S; %#okSAGROW
        fprintf('  ✓ %s %d configs, %d runs eachn', ...
            set_name, numel(S.configs), data.N_RUNS);
    else
        fprintf('  ✗ WARNING No file found for %sn', set_name);
    end
end

%% ---------------------------------------------------------------
% 2) Concatenate BATCHED sets
% ---------------------------------------------------------------
fprintf('nConcatenating batch results...n');

for i = 1numel(batched_sets)
    set_name = batched_sets{i};
    pattern = sprintf('results_%s_batch_.mat', set_name);
    batch_files = dir(pattern);

    if isempty(batch_files)
        fprintf('  ✗ WARNING No batch files found for %sn', set_name);
        continue;
    end

    % build info
    info = struct('file',{},'batch',{},'stamp',{});
    for j = 1numel(batch_files)
        fn = batch_files(j).name;
        m1 = regexp(fn, 'batch(d+)', 'tokens','once');
        m2 = regexp(fn, 'd{4}_d{2}_d{2}_d{2}_d{2}_d{2}', 'match','once');
        if ~isempty(m1)
            info(end+1).file = fn; %#okSAGROW
            info(end).batch  = str2double(m1{1});
            info(end).stamp  = m2;
        end
    end
    [~,ord] = sort([info.batch]);
    info = info(ord);

    fprintf('  Processing %s found %d batch filesn', set_name, numel(info));
    fprintf('    Merging ALL %d batches together...n', numel(info));

    % template
    data1 = load(info(1).file);
    merged = data1.set_results;
    if ~isfield(merged,'name')  isempty(merged.name), merged.name = set_name; end
    n_configs = numel(merged.configs);

    % reset run containers
    for c = 1n_configs
        merged.configs(c).all_runs.train_f1         = [];
        merged.configs(c).all_runs.test_f1          = [];
        merged.configs(c).all_runs.train_accuracy   = [];
        merged.configs(c).all_runs.test_accuracy    = [];
        merged.configs(c).all_runs.best_fitness     = [];
        merged.configs(c).all_runs.test_polygon_area= [];
        merged.configs(c).all_runs.global_run_numbers = [];
    end

    % append all batches
    for b = 1numel(info)
        data = load(info(b).file);
        B = data.set_results;
        runs_in_batch = B.configs(1).runs_in_batch;
        fprintf('      Batch %d runs %d-%d (timestamp %s)n', ...
            B.configs(1).batch_number, min(runs_in_batch), max(runs_in_batch), info(b).stamp);

        for c = 1n_configs
            merged.configs(c).all_runs.train_f1           = [merged.configs(c).all_runs.train_f1;         B.configs(c).all_runs.train_f1];
            merged.configs(c).all_runs.test_f1            = [merged.configs(c).all_runs.test_f1;          B.configs(c).all_runs.test_f1];
            merged.configs(c).all_runs.train_accuracy     = [merged.configs(c).all_runs.train_accuracy;   B.configs(c).all_runs.train_accuracy];
            merged.configs(c).all_runs.test_accuracy      = [merged.configs(c).all_runs.test_accuracy;    B.configs(c).all_runs.test_accuracy];
            merged.configs(c).all_runs.best_fitness       = [merged.configs(c).all_runs.best_fitness;     B.configs(c).all_runs.best_fitness];
            merged.configs(c).all_runs.test_polygon_area  = [merged.configs(c).all_runs.test_polygon_area;B.configs(c).all_runs.test_polygon_area];
            merged.configs(c).all_runs.global_run_numbers = [merged.configs(c).all_runs.global_run_numbers; runs_in_batch()];
        end
    end

    % recompute meansstds + total_runs_completed
    for c = 1n_configs
        cfg = merged.configs(c);
        merged.configs(c).train_f1_mean        = mean(cfg.all_runs.train_f1);
        merged.configs(c).train_f1_std         = std(cfg.all_runs.train_f1);
        merged.configs(c).test_f1_mean         = mean(cfg.all_runs.test_f1);
        merged.configs(c).test_f1_std          = std(cfg.all_runs.test_f1);
        merged.configs(c).train_accuracy_mean  = mean(cfg.all_runs.train_accuracy);
        merged.configs(c).train_accuracy_std   = std(cfg.all_runs.train_accuracy);
        merged.configs(c).test_accuracy_mean   = mean(cfg.all_runs.test_accuracy);
        merged.configs(c).test_accuracy_std    = std(cfg.all_runs.test_accuracy);
        merged.configs(c).best_fitness_mean    = mean(cfg.all_runs.best_fitness);
        merged.configs(c).best_fitness_std     = std(cfg.all_runs.best_fitness);
        merged.configs(c).test_polygon_area_mean = mean(cfg.all_runs.test_polygon_area);
        merged.configs(c).test_polygon_area_std  = std(cfg.all_runs.test_polygon_area);
        merged.configs(c).total_runs_completed = numel(cfg.all_runs.test_f1);
    end

    % best config
    [~,bi] = max([merged.configs.test_f1_mean]);
    merged.best_config = merged.configs(bi);

    % stash
    all_results{end+1} = merged; %#okSAGROW
    fprintf('    ✓ Merged %s %d configs, %d total runs eachn', ...
        set_name, n_configs, merged.configs(1).total_runs_completed);

    % save merged
    set_results = merged; %#okNASGU
    save(sprintf('results_%s_MERGED_%s.mat', set_name, timestamp_str), 'set_results');
end

function n = getRunsSafe(cfg, S)
% Robustly determine number of runs for a best_config.
    if isstruct(cfg) && isfield(cfg,'total_runs_completed') && ~isempty(cfg.total_runs_completed)
        n = double(cfg.total_runs_completed); return;
    end
    if isfield(cfg,'all_runs')
        if isfield(cfg.all_runs,'test_f1') && ~isempty(cfg.all_runs.test_f1)
            n = numel(cfg.all_runs.test_f1); return;
        elseif isfield(cfg.all_runs,'test_accuracy') && ~isempty(cfg.all_runs.test_accuracy)
            n = numel(cfg.all_runs.test_accuracy); return;
        end
    end
    if nargin=2 && isfield(S,'configs') && ~isempty(S.configs) ...
            && isfield(S.configs(1),'all_runs') && isfield(S.configs(1).all_runs,'test_f1')
        n = numel(S.configs(1).all_runs.test_f1); return;
    end
    n = NaN;
end

function total = count_total_runs(all_results)
% Robustly sum runs over every configuration in every set.
    total = 0;
    for i = 1numel(all_results)
        S = all_results{i};
        for c = 1numel(S.configs)
            cfg = S.configs(c);
            if isfield(cfg,'total_runs_completed') && ~isempty(cfg.total_runs_completed)
                total = total + double(cfg.total_runs_completed);
            elseif isfield(cfg,'all_runs') && isfield(cfg.all_runs,'test_f1') && ~isempty(cfg.all_runs.test_f1)
                total = total + numel(cfg.all_runs.test_f1);
            elseif isfield(cfg,'all_runs') && isfield(cfg.all_runs,'test_accuracy') && ~isempty(cfg.all_runs.test_accuracy)
                total = total + numel(cfg.all_runs.test_accuracy);
            end
        end
    end
    % fallback if still 0 infer per-set
    if ~isfinite(total)  total==0
        for i = 1numel(all_results)
            S = all_results{i};
            if isfield(S,'configs') && ~isempty(S.configs)
                runs_per = NaN;
                cfg1 = S.configs(1);
                if isfield(cfg1,'total_runs_completed') && ~isempty(cfg1.total_runs_completed)
                    runs_per = double(cfg1.total_runs_completed);
                elseif isfield(cfg1,'all_runs') && isfield(cfg1.all_runs,'test_f1') && ~isempty(cfg1.all_runs.test_f1)
                    runs_per = numel(cfg1.all_runs.test_f1);
                end
                if ~isnan(runs_per), total = total + numel(S.configs)runs_per; end
            end
        end
    end
end

function m = safe_mean(all_runs, field)
    if isfield(all_runs, field) && ~isempty(all_runs.(field))
        v = all_runs.(field); m = mean(v);
    else
        m = NaN;
    end
end

function s = safe_std(all_runs, field)
    if isfield(all_runs, field) && ~isempty(all_runs.(field))
        v = all_runs.(field); s = std(v);
    else
        s = NaN;
    end
end

function [m,s] = best_polygon(B)
% Return polygon area meanstd for a best_config (robust to older files)
    if isfield(B,'test_polygon_area_mean') && isfield(B,'test_polygon_area_std') ...
            && ~isempty(B.test_polygon_area_mean)
        m = B.test_polygon_area_mean; s = B.test_polygon_area_std; return;
    end
    if isfield(B,'all_runs') && isfield(B.all_runs,'test_polygon_area') ...
            && ~isempty(B.all_runs.test_polygon_area)
        m = mean(B.all_runs.test_polygon_area); s = std(B.all_runs.test_polygon_area); return;
    end
    m = NaN; s = NaN;
end
